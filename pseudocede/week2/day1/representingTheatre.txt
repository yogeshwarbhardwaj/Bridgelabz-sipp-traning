// Class representing a Ticket
class Ticket {
    private String movieName;
    private String seatNumber;
    private double price;
    private boolean isBooked;

    // Constructor
    Ticket(String movieName, String seatNumber, double price) {
        this.movieName = movieName;
        this.seatNumber = seatNumber;
        this.price = price;
        this.isBooked = false;
    }

    // Encapsulation: Getter & Setter methods
    public String getMovieName() { return movieName; }
    public String getSeatNumber() { return seatNumber; }
    public double getPrice() { return price; }
    public boolean getStatus() { return isBooked; }

    public void bookTicket() { isBooked = true; }
    public void cancelTicket() { isBooked = false; }
}
